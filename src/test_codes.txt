import streamlit as st
import pandas as pd
import seaborn as sns
import matplotlib.pyplot as plt
import traceback
from datetime import datetime

# Your modules
from sql_generator import query_ollama, conn, cursor
from fraud_detector import query
from plotting import df, generate_code

# Configure Streamlit page
st.set_page_config(page_title="Fraud Detection Chat", layout="wide")

# ---- SESSION SETUP ----
if "sessions" not in st.session_state:
    st.session_state.sessions = {}  # {session_name: [messages]}
    st.session_state.current_session = "New Chat"

if "current_session" not in st.session_state:
    st.session_state.current_session = "New Chat"

# ---- SIDEBAR SESSION SELECTOR ----
with st.sidebar:
    st.title("💬 Chat History")
    if st.button("➕ New Chat"):
        new_session_name = f"Chat {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}"
        st.session_state.sessions[new_session_name] = []
        st.session_state.current_session = new_session_name

    session_names = list(st.session_state.sessions.keys())
    if session_names:
        selected = st.radio("Select a session:", session_names, index=session_names.index(st.session_state.current_session))
        st.session_state.current_session = selected

    task_option = st.radio("🛠 Task", ["Prediction", "Plotting", "SQL Query"])

# Get messages of current session
chat_history = st.session_state.sessions.get(st.session_state.current_session, [])

# ---- MAIN CHAT INTERFACE ----
st.title("🤖 Auto Insurance Fraud Chat Assistant")

# Show chat messages
for msg in chat_history:
    with st.chat_message(msg["role"]):
        st.markdown(msg["content"])

# ---- INPUT BOX ----
prompt = st.chat_input("Enter your question...")

if prompt:
    # Save user message
    chat_history.append({"role": "user", "content": prompt})
    with st.chat_message("user"):
        st.markdown(prompt)

    # Generate response based on task
    try:
        if task_option == "Prediction":
            with st.spinner("Detecting fraud..."):
                result = query(prompt)
                response = f"**Prediction Result:**\n\n{result}"

        elif task_option == "Plotting":
            with st.spinner("Generating plot..."):
                code = generate_code(prompt, df.columns)
                plt.clf()
                local_vars = {"df": df.copy()}
                exec(code, {"pd": pd, "sns": sns, "plt": plt}, local_vars)
                st.pyplot(plt.gcf())
                plt.clf()
                response = f"**Code Executed:**\n```python\n{code}\n```"

        elif task_option == "SQL Query":
            with st.spinner("Generating SQL..."):
                sql_query = query_ollama(prompt)
                if "insurance_claims" not in sql_query.lower():
                    raise ValueError("Generated query does not target the `insurance_claims` table.")
                cursor.execute(sql_query)
                rows = cursor.fetchall()
                columns = [col[0] for col in cursor.description]
                result_df = pd.DataFrame(rows, columns=columns)
                st.dataframe(result_df)
                response = f"**Generated SQL Query:**\n```sql\n{sql_query}\n```"

    except Exception:
        response = f"❌ Error:\n```\n{traceback.format_exc()}\n```"

    # Save and show assistant message
    chat_history.append({"role": "assistant", "content": response})
    with st.chat_message("assistant"):
        st.markdown(response)

    # Save updated chat history
    st.session_state.sessions[st.session_state.current_session] = chat_history

    # Fill missing and encode

    df = pd.get_dummies(df, drop_first=True)

    target = 'fraud_reported_Y'
    X = df.drop(columns=[target])
    y = df[target]

    X_train, X_val, y_train, y_val = train_test_split(X, y, test_size=0.1, random_state=109)

    clf = DecisionTreeClassifier(max_depth=4, random_state=109)
    clf.fit(X_train, y_train)
    y_pred = clf.predict(X_val)

    #st.write("### Model Performance on Validation Set")

    #st.write(f"**Precision:** {precision_score(y_val, y_pred):.2f}")
    
    return clf, X.columns.tolist()

import streamlit as st

import requests
import json

def query(user_prompt):
    url = 'http://localhost:11434/api/generate'
    sys_prompt = '''
    You are a chatbot designed solely to provide detailed explanations for the questions posed by the user.
    Your role is not to offer direct solutions, but to clarify concepts, elaborate on ideas, and guide the user to a better understanding of their query.
    '''
    payload = {'model' : 'llama3.2', 'prompt' : user_prompt, 'system' : sys_prompt}
    try:
        response = requests.post(url = url, json = payload, headers={"Content-Type": "application/json"})
        if response.status_code == 200:
            result = response.text
            lines = result.strip().split("\n")
            full_response = "".join(json.loads(line)["response"] for line in lines)
            return full_response
        else:
            return "Error: Unable to reach the AI model service."
    except Exception as e:
        return f"Error: {str(e)}"
